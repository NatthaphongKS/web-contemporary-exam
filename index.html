Migration

PHP database/migrations/xxxx_create_books_table.php 
<?php

use Illuminate\Database\Migrations\Migration;
use Illuminate\Database\Schema\Blueprint;
use Illuminate\Support\Facades\Schema;

return new class extends Migration
{
    public function up(): void
    {
        Schema::create('books', function (Blueprint $table) {
            $table->id();
            $table->foreignId('category_id')->constrained('categories')->onDelete('cascade');
            $table->string('title');
            $table->string('author');
            $table->integer('published_year');
            $table->boolean('is_available')->default(true);
            $table->string('cover_image')->nullable();
            $table->timestamps();
        });
    }

    public function down(): void
    {
        Schema::dropIfExists('books');
    }
};

2.4 Model Eloquent: Category & Book Relations

Models

กำหนด $fillable และ Relationship แบบ 1:N (Category hasMany Books, Book belongsTo Category)

PHP app/Models/Category.php & Book.php 
// File: app/Models/Category.php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Category extends Model
{
    use HasFactory;
    protected $fillable = ['name', 'description'];

    public function books()
    {
        return $this->hasMany(Book::class);
    }
}

// File: app/Models/Book.php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Book extends Model
{
    use HasFactory;

    protected $fillable = [
        'category_id',
        'title',
        'author',
        'published_year',
        'is_available',
        'cover_image'
    ];

    protected $casts = [
        'is_available' => 'boolean',
        'published_year' => 'integer',
        'category_id' => 'integer'
    ];

    public function category()
    {
        return $this->belongsTo(Category::class);
    }
}

/*
 * Endpoints ที่ได้:
 * GET    /api/categories         -> ดึงหมวดหมู่ทั้งหมด (200 OK)
 * POST   /api/categories         -> เพิ่มหมวดหมู่ใหม่ (201 Created)

 * GET    /api/books              -> ดึงหนังสือทั้งหมด (200 OK)

3.4 BookController: CRUD & File Upload ปกหนังสือ

Controller

Controller จัดการหนังสือ รองรับการอัปโหลดไฟล์รูปภาพลง storage และคืนค่า Status Code ตามข้อกำหนด (200, 201, 404)

PHP app/Http/Controllers/Api/BookController.php 
<?php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Book;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class BookController extends Controller
{
    public function index()
    {
        $books = Book::with('category')->latest()->get()->map(function ($book) {
            $book->cover_url = $book->cover_image ? asset('storage/' . $book->cover_image) : null;
            return $book;
        });

        return response()->json($books, 200);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'category_id'    => 'required|exists:categories,id',
            'title'          => 'required|string|max:255',
            'author'         => 'required|string|max:255',
            'published_year' => 'required|integer|min:1000|max:' . (date('Y') + 1),
            'is_available'   => 'nullable|boolean',
            'cover_image'    => 'nullable|image|mimes:jpeg,png,jpg,webp|max:2048'
        ]);

        $validated['is_available'] = filter_var($request->input('is_available', true), FILTER_VALIDATE_BOOLEAN);

        // จัดการอัปโหลดไฟล์ปกหนังสือ
        if ($request->hasFile('cover_image')) {
            $path = $request->file('cover_image')->store('covers', 'public');
            $validated['cover_image'] = $path;
        }

        $book = Book::create($validated);
        $book->load('category');
        $book->cover_url = $book->cover_image ? asset('storage/' . $book->cover_image) : null;

        return response()->json($book, 201);
    }

    public function show(string $id)
    {
        $book = Book::with('category')->find($id);
        if (!$book) {
            return response()->json(['message' => 'ไม่พบข้อมูลหนังสือที่ต้องการ'], 404);
        }

        $book->cover_url = $book->cover_image ? asset('storage/' . $book->cover_image) : null;
        return response()->json($book, 200);
    }

    public function update(Request $request, string $id)
    {
        $book = Book::find($id);
        if (!$book) {
            return response()->json(['message' => 'ไม่พบข้อมูลหนังสือที่ต้องการแก้ไข'], 404);
        }

        $validated = $request->validate([
            'category_id'    => 'sometimes|required|exists:categories,id',
            'title'          => 'sometimes|required|string|max:255',
            'author'         => 'sometimes|required|string|max:255',
            'published_year' => 'sometimes|required|integer|min:1000',
            'is_available'   => 'nullable|boolean',
            'cover_image'    => 'nullable|image|mimes:jpeg,png,jpg,webp|max:2048'
        ]);

        if ($request->has('is_available')) {
            $validated['is_available'] = filter_var($request->input('is_available'), FILTER_VALIDATE_BOOLEAN);
        }

        // ถ้ามีการส่งรูปใหม่มา ให้ลบรูปเดิมแล้วบันทึกรูปใหม่
        if ($request->hasFile('cover_image')) {
            if ($book->cover_image && Storage::disk('public')->exists($book->cover_image)) {
                Storage::disk('public')->delete($book->cover_image);
            }
            $path = $request->file('cover_image')->store('covers', 'public');
            $validated['cover_image'] = $path;
        }

        $book->update($validated);
        $book->load('category');
        $book->cover_url = $book->cover_image ? asset('storage/' . $book->cover_image) : null;

        return response()->json($book, 200);
    }

    public function destroy(string $id)
    {
        $book = Book::find($id);
        if (!$book) {
            return response()->json(['message' => 'ไม่พบข้อมูลหนังสือที่ต้องการลบ'], 404);
        }

        // ลบรูปภาพออกจาก Storage
        if ($book->cover_image && Storage::disk('public')->exists($book->cover_image)) {
            Storage::disk('public')->delete($book->cover_image);
        }

        $book->delete();
        return response()->json(['message' => 'ลบข้อมูลหนังสือสำเร็จ'], 200);
    }
}

4.1 ตั้งค่า API Base URL & ดึงหมวดหมู่ใส่ Dropdown

JavaScript

ดึงข้อมูลหมวดหมู่จาก GET /api/categories มาใส่ในแท็ก <select>

JAVASCRIPT frontend/script.js 
const API_BASE = 'http://localhost:8000/api';

// ดึงหมวดหมู่มาใส่ใน Select dropdown
async function loadCategories() {
  try {
    const res = await fetch(`${API_BASE}/categories`);
    const categories = await res.json();
    
    const selectEl = document.getElementById('category_id');
    if (!selectEl) return;
    
    selectEl.innerHTML = '<option value="">-- กรุณาเลือกหมวดหมู่ --</option>';
    categories.forEach(cat => {
      selectEl.innerHTML += `<option value="${cat.id}">${cat.name}</option>`;
    });
  } catch (error) {
    console.error('Error loading categories:', error);
    alert('ไม่สามารถโหลดรายการหมวดหมู่ได้');
  }
}

4.2 ดึงรายการหนังสือทั้งหมดมาแสดงในตาราง (GET /api/books)

JavaScript

ดึงข้อมูลและ Render ลง <tbody> พร้อมรูปภาพปก สถานะยืม และปุ่มแก้ไข/ลบ

JAVASCRIPT frontend/script.js 
async function loadBooks() {
  try {
    const res = await fetch(`${API_BASE}/books`);
    const books = await res.json();
    
    const tbody = document.getElementById('bookTableBody');
    if (!tbody) return;
    
    if (books.length === 0) {
      tbody.innerHTML = '<tr><td colspan="7" style="text-align:center;">ไม่พบข้อมูลหนังสือ</td></tr>';
      return;
    }
    
    tbody.innerHTML = books.map(b => `
      <tr>
        <td>${b.id}</td>
        <td>
          ${b.cover_url 
            ? `<img src="${b.cover_url}" width="50" height="70" style="object-fit:cover; border-radius:4px;">` 
            : '<span style="color:#888;">ไม่มีปก</span>'}
        </td>
        <td><strong>${b.title}</strong></td>
        <td>${b.author}</td>
        <td>${b.category ? b.category.name : '-'}</td>
        <td>${b.published_year}</td>
        <td>
          <span class="badge ${b.is_available ? 'badge-success' : 'badge-danger'}">
            ${b.is_available ? 'พร้อมยืม' : 'ถูกยืมแล้ว'}
          </span>
        </td>
        <td>
          <button onclick="openEditModal(${b.id})" class="btn btn-sm btn-edit">แก้ไข</button>
          <button onclick="deleteBook(${b.id})" class="btn btn-sm btn-delete">ลบ</button>
        </td>
      </tr>
    `).join('');
  } catch (error) {
    console.error('Error loading books:', error);
  }
}

4.3 บันทึกข้อมูลหนังสือใหม่ (POST /api/books)

JavaScript

ดึงค่าจากฟอร์มผ่าน FormData เพื่อส่งข้อมูลและไฟล์รูปภาพไปยัง API

JAVASCRIPT frontend/script.js 
async function handleCreateBook(event) {
  event.preventDefault();
  const form = event.target;
  const formData = new FormData(form);
  
  // จัดการค่า Checkbox is_available
  const isAvailableCheckbox = form.querySelector('[name="is_available"]');
  formData.set('is_available', isAvailableCheckbox && isAvailableCheckbox.checked ? '1' : '0');

  try {
    const res = await fetch(`${API_BASE}/books`, {
      method: 'POST',
      body: formData // ห้ามใส่ Content-Type header เมื่อใช้ FormData
    });

    if (res.status === 201) {
      alert('เพิ่มหนังสือสำเร็จ!');
      form.reset();
      closeModal();
      loadBooks();
    } else {
      const errData = await res.json();
      alert('เกิดข้อผิดพลาด: ' + (errData.message || JSON.stringify(errData.errors)));
    }
  } catch (error) {
    console.error('Create error:', error);
    alert('ไม่สามารถเชื่อมต่อเซิร์ฟเวอร์ได้');
  }
}

4.4 แก้ไขข้อมูลหนังสือ (เทคนิค _method=PUT สำคัญมาก)

Laravel Pro-Tip

PHP มีข้อจำกัดไม่อ่านไฟล์ในคำขอ PUT ดังนั้นต้องส่งเป็น POST แล้วแนบ _method=PUT

JAVASCRIPT frontend/script.js 
async function handleUpdateBook(event, bookId) {
  event.preventDefault();
  const form = event.target;
  const formData = new FormData(form);
  
  // [สำคัญที่สุด] หลอก Laravel ให้ทราบว่าเป็นคำขอ PUT
  formData.append('_method', 'PUT');
  
  const isAvailableCheckbox = form.querySelector('[name="is_available"]');
  formData.set('is_available', isAvailableCheckbox && isAvailableCheckbox.checked ? '1' : '0');

  try {
    // ส่งด้วย method POST เสมอเพื่อให้อัปโหลดไฟล์สำเร็จ
    const res = await fetch(`${API_BASE}/books/${bookId}`, {
      method: 'POST',
      body: formData
    });

    if (res.status === 200) {
      alert('แก้ไขข้อมูลสำเร็จ!');
      closeModal();
      loadBooks();
    } else {
      const errData = await res.json();
      alert('เกิดข้อผิดพลาด: ' + (errData.message || JSON.stringify(errData.errors)));
    }
  } catch (error) {
    console.error('Update error:', error);
  }
}

4.5 ลบข้อมูลหนังสือ (DELETE /api/books/:id)

JavaScript

แสดง Confirm Dialog แล้วส่งคำขอลบไปยัง Backend

JAVASCRIPT frontend/script.js 
async function deleteBook(id) {
  if (!confirm(`คุณต้องการลบหนังสือรหัส ${id} ใช่หรือไม่?`)) {
    return;
  }

  try {
    const res = await fetch(`${API_BASE}/books/${id}`, {
      method: 'DELETE'
    });

    if (res.status === 200) {
      alert('ลบหนังสือสำเร็จ');
      loadBooks();
    } else if (res.status === 404) {
      alert('ไม่พบข้อมูลหนังสือเล่มนี้ในระบบ');
    } else {
      alert('ไม่สามารถลบข้อมูลได้');
    }
  } catch (error) {
    console.error('Delete error:', error);
  }
}

STEP 6 • PITFALLS & CHECKLIST

06. กับดักยอดฮิต & เช็กลิสต์ตรวจผลสอบ

วิธีแก้บั๊กฉุกเฉินที่มักเจอในห้องสอบ (CORS, Storage 404, PHP PUT Multipart, CSRF 419) พร้อมเช็กลิสต์ก่อนส่งงาน

6.1 Trap 1: ติดปัญหา CORS Policy (หน้าบ้านยิง API ไม่เข้า)

CORS Fix

เมื่อหน้าบ้านเปิดผ่านเบราว์เซอร์แล้วยิงหา API ไม่ได้ ติด CORS

PHP config/cors.php 
# สำหรับ Laravel 11 ถ้ายังไม่มี config/cors.php ให้ดึงออกมาด้วยคำสั่ง:
php artisan config:publish cors

# จากนั้นเปิดไฟล์ config/cors.php และตั้งค่าดังนี้:
return [
    'paths' => ['api/*', 'sanctum/csrf-cookie', 'storage/*'],
    'allowed_methods' => ['*'],
    'allowed_origins' => ['*'],
    'allowed_origins_patterns' => [],
    'allowed_headers' => ['*'],
    'exposed_headers' => [],
    'max_age' => 0,
    'supports_credentials' => false,
];

# รันคำสั่งเคลียร์แคช:
php artisan config:clear

6.2 Trap 2: รูปอัปโหลดติด แต่เปิดดูแล้วขึ้น 404 Not Found

Storage Fix

แก้ปัญหา Symlink ขาด หรือรูปภาพเปิดไม่ขึ้น

BASH Terminal Command 
# 1. ลบโฟลเดอร์หรือชอร์ตคัต public/storage ออกก่อน
# 2. รันคำสั่งสร้าง Symlink ใหม่:
php artisan storage:link

# 3. ตรวจสอบใน .env ให้ APP_URL=http://localhost:8000

6.3 Trap 3: ปัญหา 419 CSRF Token Mismatch

Route Fix

ตรวจสอบว่า URL ขึ้นต้นด้วย /api/ หรือไม่ เพราะ Laravel จะปิดการตรวจ CSRF เฉพาะใน routes/api.php

JAVASCRIPT API URL Tip 
// ถูกต้อง:
const url = 'http://localhost:8000/api/books';

// ผิด (จะติด 419 CSRF):
const url = 'http://localhost:8000/books';
